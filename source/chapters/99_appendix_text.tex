\chapter{Preliminary Experiments}

\section{Algorithms}\label{sec:appendix_algorithms}

\Cref{alg:training_loop,alg:random_walk,alg:threshold} summarize the proposed method in
pseudocode. \Cref{alg:training_loop} is the bidirectional training loop that alternates
between exploration on random-walk candidates and exploitation on the frontier and
solved buffers, \Cref{alg:random_walk} expands the frontier by simulating action-noise
random walks from the nearest neighbor of a sampled task-space point, and
\Cref{alg:threshold} sorts the collected rollouts into the frontier buffer $F$ and the
solved buffer $S$ based on their return.

\begin{algorithm}[H]
\caption{Bidirectional frontier training loop}\label{alg:training_loop}
\begin{algorithmic}[1]
\Require{} goal $G$, initial distribution $p_0$, task space $\mathcal{T}$
\Require{} buffer max sizes $M_F, M_S$, number of subsamples $N_s$
\Require{} $N_{\text{exploit}}, N_{\text{explore}}$ (both even), $R_{\text{min}}, R_{\text{max}}$
\Statex{}
\State{} \textbf{Initialize} $F^G, F^S$ with max size $M_F$; \quad $S^G, S^S$ with max size $M_S$
\State{} \textbf{Initialize} $\mathcal{I}^G \gets \{s_0\}, \; s_0 \sim p_0$; \quad $\mathcal{I}^S \gets \{G\}$
\Statex{}
\For{$e = 1, \ldots{}, E$}
    \State{} $C^G \gets \Call{RandomWalk}{\mathcal{I}^G, \mathcal{T}, \mathcal{I}^S, p_{\text{connect}}, N_s}$ \Comment{frontier $\xrightarrow{\text{rw}}$ expanded candidates}
    \State{} $C^S \gets \Call{RandomWalk}{\mathcal{I}^S, \mathcal{T}, \mathcal{I}^G, p_{\text{connect}}, N_s}$
    \Statex{}
    \State{} $D^G_{\text{explore}} \gets \Call{Train}{p_0, C^G, N_{\text{explore}}/2}$ \Comment{explore: train on rw candidates}
    \State{} $D^S_{\text{explore}} \gets \Call{Train}{C^S, G, N_{\text{explore}}/2}$
    \Statex{}
    \State{} $D^G_{\text{exploit}} \gets \Call{Train}{p_0, F^G \cup S^G, N_{\text{exploit}}/2}$ \Comment{exploit: train on frontier}
    \State{} $D^S_{\text{exploit}} \gets \Call{Train}{F^S \cup S^S, G, N_{\text{exploit}}/2}$ \Comment{and replay solved}
    \Statex{}
    \State{} $F^G, S^G \gets \Call{Threshold}{D^G_{\text{exploit}}, D^G_{\text{explore}}, R_{\text{min}}, R_{\text{max}}, F^G, S^G, \text{goal}}$
    \State{} $F^S, S^S \gets \Call{Threshold}{D^S_{\text{exploit}}, D^S_{\text{explore}}, R_{\text{min}}, R_{\text{max}}, F^S, S^S, \text{start}}$
\EndFor{}
\end{algorithmic}
\end{algorithm}

\begin{algorithm}[H]
\caption{Random walk frontier expansion}\label{alg:random_walk}
\begin{algorithmic}[1]
\Procedure{RandomWalk}{$\mathcal{I}, \mathcal{T}, \mathcal{I}', p_{\text{connect}}, N_s$}
    \State{} $\mathcal{C} \gets \{\,\}$ \Comment{candidate set}
    \If{$u < p_{\text{connect}}$, \quad $u \sim \mathcal{U}[0,1]$}
        \State{} $x_{\text{random}} \gets \Call{Sample}{\mathcal{I}'}$ \Comment{sample target}
    \Else{}
        \State{} $x_{\text{random}} \gets \Call{Sample}{\mathcal{T}}$ \Comment{sample task space}
    \EndIf{}
    \State{} $s_0 \gets \Call{NearestNeighbor}{x_{\text{random}}, \mathcal{I}}$
    \For{$t = 1, \ldots{}, T_B$}
        \State{} $a_t = \epsilon_t, \quad \epsilon_t \sim \mathcal{N}(0, \Sigma)$ 
        \State{} $s_t \sim P(s_t \mid s_{t-1}, a_t)$ \Comment{simulate random walk from nearest neighbor}
        \If{$t \bmod N_s = 0$}
            \State{} $\mathcal{C} \gets \mathcal{C} \cup \{s_t\}$ \Comment{subsample}
        \EndIf{}
    \EndFor{}
    \State{} \Return{} $\mathcal{C}$
\EndProcedure{}
\end{algorithmic}
\end{algorithm}

\begin{algorithm}[H]
\caption{Return-based thresholding of rollouts into frontier and solved buffers}\label{alg:threshold}
\begin{algorithmic}[1]
\Procedure{Threshold}{$D_{\text{exploit}}, D_{\text{explore}}, R_{\text{min}}, R_{\text{max}}, F, S, \text{mode}$}
    \State{} $D \gets D_{\text{exploit}} \cup D_{\text{explore}}$
    \State{} $R, \mathcal{T}\!r \gets D$ \Comment{returns and trajectories}
    \For{$(r, \tau) \in (R, \mathcal{T}\!r)$}
        \If{$\text{mode} = \text{goal}$}
            \State{} $x \gets \tau.\Call{Last}{\,}$ \Comment{extract relevant state}
        \Else{}
            \State{} $x \gets \tau.\Call{First}{\,}$
        \EndIf{}
        \If{$R_{\text{min}} < r < R_{\text{max}}$}
            \State{} $F.\Call{Append}{x}$ \Comment{threshold}
        \ElsIf{$R_{\text{max}} < r$}
            \State{} $S.\Call{Append}{x}$
        \EndIf{}
    \EndFor{}
    \State{} \Return{} $F, S$
\EndProcedure{}
\end{algorithmic}
\end{algorithm}
