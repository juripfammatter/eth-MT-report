\begin{algorithm}[H]
\caption{Random walk frontier expansion}\label{alg:random_walk}
\small
\begin{algorithmic}[1]
\Procedure{RandomWalk}{$\mathcal{I}, \mathcal{T}, S^{\text{other}}, p_{\text{connect}}, \Sigma, N_s, T_B$}
    \If{$u < p_{\text{connect}}$, \quad $u \sim \mathcal{U}[0,1]$}
        \State{} $x_{\text{rand}} \gets \phi(\Call{Sample}{S^{\text{other}}})$ \Comment{sample from other direction's solved buffer}
    \Else{}
        \State{} $x_{\text{rand}} \gets \Call{Sample}{\mathcal{T}}$ \Comment{sample uniformly from task space}
    \EndIf{}
    \State{} $s_0 \gets \Call{NearestNeighbor}{x_{\text{rand}}, \mathcal{I}}$ \Comment{w.r.t.\ $\lVert \phi(\cdot) - x_{\text{rand}} \rVert$}
    \For{$t = 1, \ldots{}, T_B$}
        \State{} $a_t = \epsilon_t, \quad \epsilon_t \sim \mathcal{N}(0, \Sigma)$
        \State{} $s_t \sim P(s_t \mid s_{t-1}, a_t)$ \Comment{simulate random walk from nearest neighbor}
    \EndFor{}
    \State{} $\mathcal{C} \gets \Call{Subsample}{\{s_1, \ldots{}, s_{T_B}\}, N_s}$ \Comment{candidate set: $N_s$ subsampled states}
    \State{} \Return{} $\mathcal{C}$
\EndProcedure{}
\end{algorithmic}
\end{algorithm}
