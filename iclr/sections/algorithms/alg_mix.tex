\begin{algorithm}[H]
\caption{Rebuild of the mixed set from frontier and solved buffers}\label{alg:mix}
\small
\begin{algorithmic}[1]
\Procedure{Mix}{$F, S, \nu, M_I$}
    \State{} $n_S \gets \min\left(\lfloor \nu \, M_I \rfloor, \; |S|\right)$ \Comment{replayed solved states}
    \State{} $n_F \gets \min\left(M_I - n_S, \; |F|\right)$ \Comment{rest from the frontier}
    \State{} $\mathcal{I} \gets \Call{Subsample}{S, n_S} \cup \Call{Subsample}{F, n_F}$ \Comment{uniform, without replacement}
    \State{} \Return{} $\mathcal{I}$
\EndProcedure{}
\end{algorithmic}
\end{algorithm}
