\subsection{Assumptions}\label{sec:assumptions}
\begin{itemize}
    \item access to uniform samples from a task space (subspace of state space)
    \item access to random walk rollouts in the state space
    \item reset agent to arbitrary states in the state space
    \item at least one known start state and one known goal state
\end{itemize}
